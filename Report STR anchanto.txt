SELECT 
	GP_ETABLISSEMENT AS SENDER_SITE,
	GP_ETABLISSDEST AS RECEIVE_SITE,
	ARTICLE_COMPL.GA_ARTICLE AS GA_ARTICLE,
	ARTICLE_COMPL.GA_CODEARTICLE AS Article,
	ARTICLE_COMPL.GA_CODEBARRE AS SKU,
	ARTICLE_COMPL.GA_LIBELLE AS articledesc,
	GCLIGNEARTDIM.gp_refinterne AS internalReference,
	GCLIGNEARTDIM.GP_DATEPIECE AS Gi_posting_date,
	sum(ISNULL(GCLIGNEARTDIM.gl_qtefact, 0)) AS Gi_qty,
	ARTICLE_COMPL.GA_PMRP AS Gi_Unit_COGS,
	ARTICLE_COMPL.GA_PVTTC AS Gi_Unit_TagPrice
FROM 
	dbo.GCLIGNEARTDIM GCLIGNEARTDIM
	INNER JOIN ARTICLE_COMPL ARTICLE_COMPL ON GCLIGNEARTDIM.GL_ARTICLE = ARTICLE_COMPL.GA_ARTICLE
WHERE 
	RTRIM(GCLIGNEARTDIM.GP_SUPPRIME)<>'X' 
	AND RTRIM(GCLIGNEARTDIM.GP_NATUREPIECEG) = 'Tem' 
	AND RTRIM(GP_SUPPRIME) <> 'X'
	AND RTRIM(ARTICLE_COMPL.GA_TYPEARTICLE) NOT IN ('FI')
	--AND GP_ETABLISSEMENT = '1399' 
	AND GP_ETABLISSDEST = '1399'
	AND GP_DATEPIECE between '2026-04-01' and '2026-04-30'
GROUP BY
	GCLIGNEARTDIM.GP_ETABLISSEMENT,
	GCLIGNEARTDIM.GP_ETABLISSDEST,
	ARTICLE_COMPL.GA_ARTICLE, 
	ARTICLE_COMPL.GA_CODEARTICLE, 
	ARTICLE_COMPL.GA_CODEBARRE, 
	ARTICLE_COMPL.GA_LIBELLE, 
	GCLIGNEARTDIM.GP_DATEPIECE, 
	GCLIGNEARTDIM.gp_refinterne, 
	ARTICLE_COMPL.GA_PMRP, 
	ARTICLE_COMPL.GA_PVTTC
