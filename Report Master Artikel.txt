    color = DI1.GDI_LIBELLE,

    Type = PRODUCT_TYPE.CC_LIBELLE,
    Category = PRODUCTCATEGORY.CC_LIBELLE,
    SubCategory = PRODUCTSUBCATEGORY.CC_LIBELLE,

    Theme = THEME.YX_LIBELLE,
    FIT = FIT.YX_LIBELLE,
    Finishing = Finishing.YX_LIBELLE,

    technology1 = Technology.YX_LIBELLE,

    season = Season.YX_LIBELLE, 
    season_year = SeasonYear.YX_LIBELLE,
    Board = Board.YX_LIBELLE,

    base_price = ARTICLE.GA_PVTTC,
    retailprice = DELAMI_SAP.[dbo].[fn_getArticlePrice](ARTICLE.GA_ARTICLE, GETDATE()),

    closed = CASE 
                WHEN ARTICLE.GA_FERME = 'X' THEN 'Y' 
                ELSE 'N' 
             END

FROM ARTICLE

JOIN ARTICLECOMPL
    ON ARTICLECOMPL.GA2_ARTICLE = ARTICLE.GA_ARTICLE

-- MASTER LOOKUP
LEFT JOIN CHOIXCOD BRAND
    ON BRAND.CC_TYPE = 'GDM'
    AND BRAND.CC_CODE = ARTICLE.GA_DOMAINE

LEFT JOIN CHOIXCOD GENDER
    ON GENDER.CC_TYPE = 'FN1'
    AND GENDER.CC_CODE = ARTICLE.GA_FAMILLENIV1

LEFT JOIN CHOIXCOD LIFECYCLE
    ON LIFECYCLE.CC_TYPE = 'FN2'
    AND LIFECYCLE.CC_CODE = ARTICLE.GA_FAMILLENIV2

LEFT JOIN CHOIXCOD PRODUCT_TYPE
    ON PRODUCT_TYPE.CC_TYPE = 'FN3'
    AND PRODUCT_TYPE.CC_CODE = ARTICLE.GA_FAMILLENIV3

LEFT JOIN CHOIXCOD PRODUCTCATEGORY
    ON PRODUCTCATEGORY.CC_TYPE = 'FN4'
    AND PRODUCTCATEGORY.CC_CODE = ARTICLECOMPL.GA2_FAMILLENIV4

LEFT JOIN CHOIXCOD PRODUCTSUBCATEGORY
    ON PRODUCTSUBCATEGORY.CC_TYPE = 'FN5'
    AND PRODUCTSUBCATEGORY.CC_CODE = ARTICLECOMPL.GA2_FAMILLENIV5

-- EXTENSION
LEFT JOIN CHOIXEXT THEME 
    ON THEME.YX_TYPE = 'LA1' 
    AND THEME.YX_CODE = ARTICLE.GA_LIBREART1

LEFT JOIN CHOIXEXT FIT 
    ON FIT.YX_TYPE = 'LA2' 
    AND FIT.YX_CODE = ARTICLE.GA_LIBREART2

LEFT JOIN CHOIXEXT Finishing 
    ON Finishing.YX_TYPE = 'LA3' 
    AND Finishing.YX_CODE = ARTICLE.GA_LIBREART3

LEFT JOIN CHOIXEXT Fabric
    ON Fabric.YX_TYPE = 'LA4' 
    AND Fabric.YX_CODE = ARTICLE.GA_LIBREART4

LEFT JOIN CHOIXEXT Technology
    ON Technology.YX_TYPE = 'LA5' 
    AND Technology.YX_CODE = ARTICLE.GA_LIBREART5

LEFT JOIN CHOIXEXT Season
    ON Season.YX_TYPE = 'LA7' 
    AND Season.YX_CODE = ARTICLE.GA_LIBREART7

LEFT JOIN CHOIXEXT SeasonYear
    ON SeasonYear.YX_TYPE = 'LA8' 
    AND SeasonYear.YX_CODE = ARTICLE.GA_LIBREART8

LEFT JOIN CHOIXEXT Board
    ON Board.YX_TYPE = 'LA9' 
    AND Board.YX_CODE = ARTICLE.GA_LIBREART9

-- DIMENSION
LEFT JOIN DIMENSION DI1
    ON LTRIM(RTRIM(DI1.GDI_TYPEDIM)) = 'DI1'
    AND DI1.GDI_CODEDIM = ARTICLE.GA_CODEDIM1
    AND DI1.GDI_GRILLEDIM = ARTICLE.GA_GRILLEDIM1

LEFT JOIN DIMENSION DI2
    ON LTRIM(RTRIM(DI2.GDI_TYPEDIM)) = 'DI2'
    AND DI2.GDI_CODEDIM = ARTICLE.GA_CODEDIM2
    AND DI2.GDI_GRILLEDIM = ARTICLE.GA_GRILLEDIM2

WHERE ARTICLE.GA_STATUTART <> 'GEN'
