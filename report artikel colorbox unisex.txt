SELECT 
    A.GA_CODEARTICLE   AS item_code,
    A.GA_CODEBARRE     AS barcode,
    A.GA_LIBELLE       AS description,

    -- COLOR
    ISNULL(DI1.GDI_LIBELLE, '') AS color,

    -- SIZE
    ISNULL(DI2.GDI_LIBELLE, '') AS size,

    -- GENDER
    GENDER.CC_LIBELLE  AS gender

FROM ARTICLE A

JOIN ARTICLECOMPL AC
    ON AC.GA2_ARTICLE = A.GA_ARTICLE

-- COLOR
LEFT JOIN DIMENSION DI1
    ON LTRIM(RTRIM(DI1.GDI_TYPEDIM)) = 'DI1'
    AND DI1.GDI_CODEDIM = A.GA_CODEDIM1
    AND DI1.GDI_GRILLEDIM = A.GA_GRILLEDIM1

-- SIZE
LEFT JOIN DIMENSION DI2
    ON LTRIM(RTRIM(DI2.GDI_TYPEDIM)) = 'DI2'
    AND DI2.GDI_CODEDIM = A.GA_CODEDIM2
    AND DI2.GDI_GRILLEDIM = A.GA_GRILLEDIM2

-- GENDER (FN1)
LEFT JOIN CHOIXCOD GENDER
    ON GENDER.CC_TYPE = 'FN1'
    AND GENDER.CC_CODE = A.GA_FAMILLENIV1

WHERE 
    A.GA_STATUTART <> 'GEN'
    AND ISNULL(A.GA_FERME,'') <> 'X'
    AND GENDER.CC_LIBELLE = 'UNISEX'
