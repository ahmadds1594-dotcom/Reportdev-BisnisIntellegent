SELECT top 100
    P.GP_NATUREPIECEG,
    SUM(L.GL_QTEFACT) AS QTY
FROM PIECE P
JOIN LIGNE L
  ON P.GP_NATUREPIECEG = L.GL_NATUREPIECEG
 AND P.GP_SOUCHE = L.GL_SOUCHE
 AND P.GP_NUMERO = L.GL_NUMERO
 AND P.GP_INDICEG = L.GL_INDICEG
JOIN ARTICLEDIM_MODE A
  ON A.GA_ARTICLE = L.GL_ARTICLE
JOIN ETABLISS E
  ON E.ET_ETABLISSEMENT = P.GP_ETABLISSEMENT
LEFT JOIN CHOIXEXT CE
  ON CE.YX_TYPE = 'EL9'
 AND CE.YX_CODE = E.ET_LIBREET9
WHERE
    P.GP_NATUREPIECEG IN ('FFO','AVS')
    AND A.GA2_FAMILLENIV4 = 'PPM'
    AND CE.YX_LIBELLE = 'Store'
    AND P.GP_DATEPIECE >= '2025-11-01'
    AND P.GP_DATEPIECE <  '2025-11-30'
    AND L.GL_SOUCHE = '2000'
GROUP BY
    P.GP_NATUREPIECEG;
