SELECT top 100
    P.GP_NATUREPIECEG,
    SUM(L.GL_QTEFACT) AS QTY
FROM PIECE P
JOIN LIGNE L
  ON P.GP_NATUREPIECEG = L.GL_NATUREPIECEG
 AND P.GP_SOUCHE = L.GL_SOUCHE
 AND P.GP_NUMERO = L.GL_NUMERO
 AND P.GP_INDICEG = L.GL_INDICEG
JOIN ARTICLEDIM_MODE A
  ON A.GA_ARTICLE = L.GL_ARTICLE
JOIN ETABLISS E
  ON E.ET_ETABLISSEMENT = P.GP_ETABLISSEMENT
LEFT JOIN CHOIXEXT CE
  ON CE.YX_TYPE = 'EL9'
 AND CE.YX_CODE = E.ET_LIBREET9
WHERE
    P.GP_NATUREPIECEG IN ('FFO','AVS')
    AND A.GA2_FAMILLENIV4 = 'PPM'
    AND CE.YX_LIBELLE = 'Store'
    AND P.GP_DATEPIECE >= '2025-11-01'
    AND P.GP_DATEPIECE <  '2025-11-30'
    AND L.GL_SOUCHE = '2000'
GROUP BY
    P.GP_NATUREPIECEG;

SELECT 
    barcode = A.GA_CODEBARRE ,       
    item_code = A.GA_CODEARTICLE,    
    Article = A.GA_LIBELLE,     

    color = DI1.GDI_LIBELLE,     
    color_code = A.GA_CODEDIM1,

    Theme = THEME.YX_LIBELLE,
    FIT = FIT.YX_LIBELLE,
    Finishing = Finishing.YX_LIBELLE,
    Fabric_design = Fabric.YX_LIBELLE,
    technology1 = Technology.YX_LIBELLE,

    season = Season.YX_LIBELLE, 
    season_year = SeasonYear.YX_LIBELLE,
    Board = Board.YX_LIBELLE

FROM ARTICLE A

LEFT JOIN ARTICLECOMPL AC
    ON AC.GA2_ARTICLE = A.GA_ARTICLE

LEFT JOIN DIMENSION DI1
    ON DI1.GDI_TYPEDIM = 'DI1'
    AND DI1.GDI_CODEDIM = A.GA_CODEDIM1
    AND DI1.GDI_GRILLEDIM = A.GA_GRILLEDIM1


LEFT JOIN CHOIXEXT THEME 
    ON THEME.YX_TYPE = 'LA1' 
    AND THEME.YX_CODE = A.GA_LIBREART1

LEFT JOIN CHOIXEXT FIT 
    ON FIT.YX_TYPE = 'LA2' 
    AND FIT.YX_CODE = A.GA_LIBREART2

LEFT JOIN CHOIXEXT Finishing 
    ON Finishing.YX_TYPE = 'LA3' 
    AND Finishing.YX_CODE = A.GA_LIBREART3

LEFT JOIN CHOIXEXT Fabric
    ON Fabric.YX_TYPE = 'LA4' 
    AND Fabric.YX_CODE = A.GA_LIBREART4

LEFT JOIN CHOIXEXT Technology
    ON Technology.YX_TYPE = 'LA5' 
    AND Technology.YX_CODE = A.GA_LIBREART5

LEFT JOIN CHOIXEXT Season
    ON Season.YX_TYPE = 'LA7' 
    AND Season.YX_CODE = A.GA_LIBREART7

LEFT JOIN CHOIXEXT SeasonYear
    ON SeasonYear.YX_TYPE = 'LA8' 
    AND SeasonYear.YX_CODE = A.GA_LIBREART8

LEFT JOIN CHOIXEXT Board
    ON Board.YX_TYPE = 'LA9' 
    AND Board.YX_CODE = A.GA_LIBREART9

WHERE A.GA_CODEBARRE = '9990000000043'
