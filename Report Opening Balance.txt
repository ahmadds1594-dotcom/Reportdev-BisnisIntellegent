select
Site,
BARCODE,
OpeningStock		= SUM (OpeningStock),
OpeningInTransit	= SUM (OpeningInTransit),
ClosingStock		= SUM (ClosingStock),
OpeningStockAct		= SUM (OpeningStockAccounting ),
OpeningValueAct		= SUM (OpeningAccountingValue),	
ClosingStockAct		= SUM (ClosingStockAccounting), 
ClosingValAct		= SUM (ClosingValueAccounting) 
from InventoryMovement
where FromDate = '01 Dec 2025'
and site = '1209'
GROUP BY 
SITE,
BARCODE
ORDER BY SITE,
BARCODE;
