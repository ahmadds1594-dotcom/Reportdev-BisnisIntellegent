    V.T_TIERS        AS vendor_code,
    V.T_LIBELLE      AS vendor_name,
    V.T_ADRESSE1     AS address1,

    V.T_CODEPOSTAL   AS postal_code,
    V.T_MODEREGLE    AS payment_term,
    V.T_DEVISE       AS currency,

    V.YTC_TABLELIBREFOU1,
    CE.YX_LIBELLE    AS vendor_type

FROM GCVTIERSCOMPL V

    V.T_NATUREAUXI = 'FOU'

)
